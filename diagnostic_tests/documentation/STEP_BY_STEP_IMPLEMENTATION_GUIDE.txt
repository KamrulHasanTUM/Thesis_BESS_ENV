    action_magnitude = np.mean(np.abs(env._last_action))

        total_reward += inaction_penalty
        reward_breakdown['inaction_penalty'] = inaction_penalty

    # ============================================================
    # IMPROVEMENT 2: Action Utilization Bonus
    # ============================================================
    # Reward using BESS capacity (scaled actions are -1 to 1)
    utilization_bonus = 5.0 * action_magnitude
    total_reward += utilization_bonus
    reward_breakdown['utilization_bonus'] = utilization_bonus

    # ============================================================
    # IMPROVEMENT 3: Diversity Bonus
    # ============================================================
    # Count how many BESS units are actively used
    num_active = np.sum(np.abs(env._last_action) > 0.01)
    diversity_bonus = 2.0 * num_active
    total_reward += diversity_bonus
    reward_breakdown['diversity_bonus'] = diversity_bonus
    reward_breakdown['num_active_units'] = int(num_active)

    # ============================================================
    # IMPROVEMENT 4: Stuck State Penalty
    # ============================================================
    # Additional penalty for being stuck at initial SoC with no action
    num_stuck_at_initial = np.sum(
        (np.abs(env.bess_soc - 0.5) < 0.01) &
        (np.abs(env._last_action) < 0.01)
    )
    if num_stuck_at_initial > 0:
        stuck_penalty = -20.0 * num_stuck_at_initial
        total_reward += stuck_penalty
        reward_breakdown['stuck_penalty'] = stuck_penalty
